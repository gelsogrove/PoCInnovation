function myFunction(a: number, b: number): Promise<boolean> {
    return new Promise<boolean>((resolve, reject) => {
        if (a > b) {
            resolve(true);
        } else {
            reject(new Error('a is not greater than b'));
        }
    });
}



const myFunction = (a: number, b: number): Promise<boolean> => {
    return new Promise<boolean>((resolve, reject) => {
        if (a > b) {
            resolve(true);
        } else {
            reject(new Error('a is not greater than b'));
        }
    });
};


async function call() {
    try {
        const x: boolean = await myFunction(4, 10);
        console.log(x); // Stampa il valore di x (true o false)
    } catch (error) {
        console.error('Error:', error);
    }
}

================= TRY CATCH ==============================

function divideNumbers(a: number, b: number): number {
    try {
        if (b === 0) {
            throw new Error("Division by zero is not allowed.");
        }
        return a / b;
    } catch (error) {
        console.error("Error:", error.message);
        return NaN; // Restituisce NaN se si verifica un errore
    }
}

const result = divideNumbers(10, 0);
console.log(result); // Output: Error: Division by zero is not allowed. / NaN

===================== Optional Chaining =======================

let user = {
    name: 'Alice',
    address: {
        street: '123 Main St',
        city: 'Wonderland'
    }
};

// Accesso sicuro con optional chaining
let street = user?.address?.street;
console.log(street); // '123 Main St'

// Accesso a una proprietà che potrebbe non esistere
let phone = user?.phone;
console.log(phone); // undefined

// Optional Chaining +  NULLISH COALISCING
let street = user?.address?.street ?? 'No Street Provided';
let city = user?.address?.city ?? 'No City Provided';
